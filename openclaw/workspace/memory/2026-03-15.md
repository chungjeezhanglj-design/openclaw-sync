### MapLibre GL 配置要点
```javascript
const map = new maplibregl.Map({
    container: 'map',
    style: {
        version: 8,
        sources: {
            'shortbread': {
                type: 'vector',
                tiles: ['http://localhost:8081/{z}/{x}/{y}.pbf'],
                minzoom: 0,
                maxzoom: 14,
                scheme: 'xyz'  // 重要！
            }
        },
        layers: [
            { id: 'bg', type: 'background', paint: { 'background-color': '#aad5ff' } },
            { 
                id: 'land', 
                type: 'fill', 
                source: 'shortbread', 
                'source-layer': 'land',
                minzoom: 7,  // land 图层从 zoom 7 开始有数据
                paint: { 'fill-color': '#e0d8c8' } 
            }
        ]
    },
    center: [104.11, 33.96],
    zoom: 7  // 初始 zoom 需≥7 才能看到陆地
});
```
